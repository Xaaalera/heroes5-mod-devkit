cmake_minimum_required(VERSION 3.21)
project(UniversePlayerLaunchChecks LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Native launch checks require MSVC x86.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(player_launch_test player_launch_test.cpp)
target_compile_definitions(player_launch_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(player_launch_test PRIVATE bcrypt comdlg32 dxguid)
enable_testing()
add_test(NAME player_launch_boundaries COMMAND player_launch_test)
add_library(heroes5_mod_loader SHARED mod_loader.cpp)
set_target_properties(heroes5_mod_loader PROPERTIES OUTPUT_NAME "dinput8")
target_compile_definitions(heroes5_mod_loader PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(heroes5_mod_loader PRIVATE bcrypt user32)
target_link_options(heroes5_mod_loader PRIVATE /Brepro)
add_test(NAME directinput_forwarding COMMAND player_launch_test $<TARGET_FILE:heroes5_mod_loader>)
install(TARGETS heroes5_mod_loader RUNTIME DESTINATION bin)
