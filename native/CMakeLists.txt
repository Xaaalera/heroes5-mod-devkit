cmake_minimum_required(VERSION 3.21)
project(UniversePlayerLaunchChecks LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Native launch checks require MSVC x86.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(xkit_graphics_proxy SHARED graphics_proxy.cpp graphics_proxy.def)
set_target_properties(xkit_graphics_proxy PROPERTIES OUTPUT_NAME "d3d9")
target_compile_definitions(xkit_graphics_proxy PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(xkit_graphics_proxy PRIVATE bcrypt)
target_link_options(xkit_graphics_proxy PRIVATE /BASE:0x68000000 /DYNAMICBASE:NO /FIXED:NO /Brepro)
set(graphics_identity_directory "${CMAKE_CURRENT_BINARY_DIR}/graphics-identity/$<CONFIG>")
set(graphics_identity_header "${graphics_identity_directory}/graphics_build_identity.hpp")
add_custom_command(OUTPUT "${graphics_identity_header}"
    COMMAND "${CMAKE_COMMAND}" -E make_directory "${graphics_identity_directory}"
    COMMAND "${CMAKE_COMMAND}" "-DGRAPHICS_FILE=$<TARGET_FILE:xkit_graphics_proxy>"
        "-DOUTPUT_FILE=${graphics_identity_header}" -P "${CMAKE_CURRENT_SOURCE_DIR}/graphics_identity.cmake"
    DEPENDS xkit_graphics_proxy graphics_identity.cmake VERBATIM)
add_custom_target(xkit_graphics_identity DEPENDS "${graphics_identity_header}")
add_executable(graphics_proxy_test graphics_proxy_test.cpp)
target_link_libraries(graphics_proxy_test PRIVATE bcrypt)
add_library(graphics_callout_fixture SHARED graphics_proxy_test.cpp)
target_compile_definitions(graphics_callout_fixture PRIVATE XKIT_GRAPHICS_CALLOUT_FIXTURE)
add_test(NAME graphics_proxy_guard COMMAND graphics_proxy_test
    $<TARGET_FILE:xkit_graphics_proxy> $<TARGET_FILE:graphics_callout_fixture>)
set_tests_properties(graphics_proxy_guard PROPERTIES TIMEOUT 10)
option(XALKIT_BUILD_CONSOLE "Build the automatically attached development console" OFF)
if(XALKIT_BUILD_CONSOLE)
    include(console_module.cmake)
    add_dependencies(xalkit_console xkit_graphics_identity)
    target_include_directories(xalkit_console PRIVATE "${graphics_identity_directory}")
endif()
add_executable(sdk_crash_fixture crash_fixture.cpp)
add_executable(sdk_launch_gate sdk_launch_gate.cpp)
target_compile_definitions(sdk_launch_gate PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(sdk_launch_gate PRIVATE bcrypt comdlg32)
add_executable(player_launch_test player_launch_test.cpp)
target_compile_definitions(player_launch_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(player_launch_test PRIVATE bcrypt comdlg32 dxguid)
enable_testing()
add_test(NAME player_launch_boundaries COMMAND player_launch_test)
add_library(heroes5_mod_loader SHARED mod_loader.cpp)
set_target_properties(heroes5_mod_loader PROPERTIES OUTPUT_NAME "dinput8")
target_compile_definitions(heroes5_mod_loader PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(heroes5_mod_loader PRIVATE bcrypt user32)
target_link_options(heroes5_mod_loader PRIVATE /Brepro)
add_test(NAME directinput_forwarding COMMAND player_launch_test $<TARGET_FILE:heroes5_mod_loader>)
install(TARGETS heroes5_mod_loader RUNTIME DESTINATION bin)

set(animation_import_library "${CMAKE_CURRENT_BINARY_DIR}/animation_import.lib")
add_custom_command(OUTPUT "${animation_import_library}"
    COMMAND "${CMAKE_AR}" /nologo /machine:x86
        "/def:${CMAKE_CURRENT_SOURCE_DIR}/animation_import.def"
        "/out:${animation_import_library}"
    DEPENDS animation_import.def VERBATIM)
add_library(heroes5_animation_preload SHARED animation_preload.cpp animation_preload.def "${animation_import_library}")
set_target_properties(heroes5_animation_preload PROPERTIES OUTPUT_NAME "wsock32")
target_compile_definitions(heroes5_animation_preload PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(heroes5_animation_preload PRIVATE "${animation_import_library}" ws2_32 mswsock)
target_link_options(heroes5_animation_preload PRIVATE /BASE:0x6a000000 /DYNAMICBASE:NO /FIXED:NO /Brepro)
install(TARGETS heroes5_animation_preload RUNTIME DESTINATION bin)
add_library(animation_preload_fixture SHARED animation_preload_fixture.cpp)
set_target_properties(animation_preload_fixture PROPERTIES OUTPUT_NAME "granny2"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
add_executable(animation_preload_test animation_preload_test.cpp)
target_compile_definitions(animation_preload_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(animation_preload_test PRIVATE ws2_32)
add_test(NAME animation_preload_forwarding COMMAND animation_preload_test
    $<TARGET_FILE:heroes5_animation_preload> $<TARGET_FILE:animation_preload_fixture>)

add_executable(plugin_runtime_test plugin_runtime_test.cpp)
add_library(heroes5_plugin_bridge SHARED plugin_bridge.cpp)
add_library(heroes5_plugin_bridge_v2 SHARED plugin_bridge.cpp)
target_compile_definitions(heroes5_plugin_bridge_v2 PRIVATE H5_CORE_VERSION=2)
target_link_libraries(heroes5_plugin_bridge_v2 PRIVATE user32 bcrypt)
add_library(heroes5_plugin_bridge_bad_restore SHARED plugin_bridge.cpp)
target_compile_definitions(heroes5_plugin_bridge_bad_restore PRIVATE H5_CORE_VERSION=2 H5_CORE_REJECT_RESTORE=1)
target_link_libraries(heroes5_plugin_bridge_bad_restore PRIVATE user32 bcrypt)
target_link_libraries(heroes5_plugin_bridge PRIVATE user32 bcrypt)
add_executable(plugin_bridge_client plugin_bridge_client.cpp)
target_compile_definitions(plugin_bridge_client PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(plugin_bridge_client PRIVATE bcrypt comdlg32 dxguid)
foreach(target IN ITEMS heroes5_mod_loader plugin_bridge_client sdk_launch_gate)
    add_dependencies(${target} xkit_graphics_identity)
    target_include_directories(${target} PRIVATE "${graphics_identity_directory}")
endforeach()
foreach(payload IN ITEMS v1 v2 bad_abi bad_state)
    add_library(plugin_fixture_${payload} SHARED plugin_runtime_fixture.cpp)
endforeach()
target_compile_definitions(plugin_fixture_v2 PRIVATE FIXTURE_VERSION=2)
target_compile_definitions(plugin_fixture_bad_abi PRIVATE FIXTURE_ABI=999)
target_compile_definitions(plugin_fixture_bad_state PRIVATE FIXTURE_SCHEMA=999)
add_test(NAME plugin_runtime_lifecycle COMMAND plugin_runtime_test
    $<TARGET_FILE:plugin_fixture_v1> $<TARGET_FILE:plugin_fixture_v2>
    $<TARGET_FILE:plugin_fixture_bad_abi> $<TARGET_FILE:plugin_fixture_bad_state>)
add_executable(diagnostic_bus_test diagnostic_bus_test.cpp)
add_test(NAME diagnostic_bus_shared_modules COMMAND diagnostic_bus_test
    $<TARGET_FILE:plugin_fixture_v1> $<TARGET_FILE:plugin_fixture_v2>)
add_executable(plugin_bridge_test plugin_bridge_test.cpp)
target_link_libraries(plugin_bridge_test PRIVATE user32 bcrypt)
add_test(NAME plugin_bridge_machine_state COMMAND plugin_bridge_test $<TARGET_FILE:plugin_fixture_v1>)
set_tests_properties(plugin_runtime_lifecycle plugin_bridge_machine_state PROPERTIES TIMEOUT 20)
